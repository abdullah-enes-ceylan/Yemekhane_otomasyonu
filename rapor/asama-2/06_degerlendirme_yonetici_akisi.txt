# 3) Menü oluşturma: geçerli, aynı anahtarla ikinci kez, geçmiş tarih, yemeksiz
> curl -s -b yonetici -X POST -d '{"yemekhane_id":2,"tarih":"2026-11-26","ogun_id":2,"ogrenci_fiyati":"40","personel_fiyati":"90","yemek_idleri":[3,12,22]}' http://localhost:8080/api/admin/menuler
{"menuId":88} [HTTP 201]

> curl -s -b yonetici -X POST -d '{... ayni yemekhane, tarih, ogun ...}' http://localhost:8080/api/admin/menuler
{"mesaj":"Bu yemekhane, tarih ve öğün için zaten bir menü var; onu düzenleyin."} [HTTP 409]

> curl -s -b yonetici -X POST -d '{"tarih":"2026-11-01", ...}' http://localhost:8080/api/admin/menuler
{"mesaj":"Geçmiş bir tarih için menü oluşturulamaz."} [HTTP 422]

> curl -s -b yonetici -X POST -d '{..., "yemek_idleri":[]}' http://localhost:8080/api/admin/menuler
{"mesaj":"Yemekler en az bir seçim içermelidir."} [HTTP 400]

# 4) Turnike (sp_turnike_gecis) → rezervasyon 'kullanildi' → hemen değerlendirilebilir
> curl -s -b personel -X POST -d '{"menu_id":56,"puan":4}' http://localhost:8080/api/degerlendirmeler   (henüz kullanılmadı)
{"mesaj":"Yalnızca kullandığınız (turnikeden geçtiğiniz) menüleri değerlendirebilirsiniz."} [HTTP 422]

> curl -s -b yonetici -X POST -d '{"kullanici":"murat.kaya@kampus.edu.tr","yemekhane_id":1,"zaman":"2026-11-20T12:30"}' http://localhost:8080/api/admin/turnike
{"kullanici":"Murat Kaya","rezervasyonId":370,"menuId":56,"ogun":"Öğle"} [HTTP 200]

> curl -s -b yonetici -X POST -d '{... aynı geçiş tekrar ...}' http://localhost:8080/api/admin/turnike
{"mesaj":"Bu rezervasyon zaten kullanıldı."} [HTTP 422]

> curl -s -b personel -X POST -d '{"menu_id":56,"puan":4}' http://localhost:8080/api/degerlendirmeler
{"degerlendirmeId":77} [HTTP 201]

> curl -s -b personel -X POST -d '{"menu_id":56,"puan":5}' http://localhost:8080/api/degerlendirmeler   (ikinci oy)
{"mesaj":"Bu menüyü zaten değerlendirdiniz."} [HTTP 409]

> curl -s -b personel -X POST -d '{"menu_id":56,"puan":7}' http://localhost:8080/api/degerlendirmeler
{"mesaj":"Puan 1 ile 5 arasında bir tamsayı olmalıdır."} [HTTP 400]
